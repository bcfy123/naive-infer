cmake_minimum_required(VERSION 3.16)
project(naive_infer)
set(CMAKE_CXX_STANDARD 17)
find_package(Armadillo REQUIRED)
find_package(OpenMP REQUIRED)
find_package(BLAS REQUIRED)
find_package(LAPACK REQUIRED)
find_package(glog REQUIRED)
find_package(GTest REQUIRED)

set(link_lib glog::glog GTest::gtest)
if (!WIN32)
    set(link_lib "${link_lib} pthread")
endif ()
set(link_math_lib ${ARMADILLO_LIBRARIES} ${BLAS_LIBRARIES} ${LAPACK_LIBRARIES})
aux_source_directory(./test DIR_TEST)
aux_source_directory(./source DIR_SOURCE)
aux_source_directory(./source/layer/abstract DIR_ABSTRACT_LAYER)
aux_source_directory(source/layer/detail DIR_DETAIL_LAYER)

add_executable(naive_infer main.cpp ${DIR_TEST} ${DIR_SOURCE} ${DIR_ABSTRACT_LAYER} ${DIR_DETAIL_LAYER})
target_include_directories(naive_infer PUBLIC ${glog_INCLUDE_DIR})
target_include_directories(naive_infer PUBLIC ${GTest_INCLUDE_DIR})
target_include_directories(naive_infer PUBLIC ${Armadillo_INCLUDE_DIR})
target_link_libraries(naive_infer ${link_lib} ${link_math_lib} OpenMP::OpenMP_CXX)

target_include_directories(naive_infer PUBLIC ./include)
